-- Add new columns to the 'orders' table if they don't exist
ALTER TABLE public.orders
ADD COLUMN IF NOT EXISTS order_number TEXT,
ADD COLUMN IF NOT EXISTS payment_status TEXT,
ADD COLUMN IF NOT EXISTS payment_amount NUMERIC;

-- Backfill existing orders with a generated order_number if it's null
-- This assumes 'id' is a UUID. We'll use a prefix and a short hash of the UUID.
-- This is a non-destructive backfill.
UPDATE public.orders
SET order_number = 'MB-' || TO_CHAR(created_at, 'YYYYMMDD') || '-' || SUBSTRING(REPLACE(id::text, '-', ''), 1, 8)
WHERE order_number IS NULL;

-- Backfill payment_status from meta if available and payment_status is null
-- This is a non-destructive backfill.
UPDATE public.orders
SET payment_status = COALESCE(payment_status, meta->>'payment_status')
WHERE payment_status IS NULL AND meta->>'payment_status' IS NOT NULL;

-- Backfill payment_amount from meta if available and payment_amount is null
-- This is a non-destructive backfill.
UPDATE public.orders
SET payment_amount = COALESCE(payment_amount, (meta->>'total')::numeric)
WHERE payment_amount IS NULL AND meta->>'total' IS NOT NULL;

-- Create or replace the adaptive view public.deliveries_for_admin
-- This view normalizes delivery data for the admin panel.
CREATE OR REPLACE VIEW public.deliveries_for_admin AS
SELECT
    o.id AS order_id,
    COALESCE(o.order_number, 'MB-' || TO_CHAR(o.created_at, 'YYYYMMDD') || '-' || SUBSTRING(REPLACE(o.id::text, '-', ''), 1, 8)) AS order_number,
    o.status,
    o.assigned_to AS driver_id,
    d.name AS driver_name,
    COALESCE(o.payment_method, o.meta->>'payment_method') AS payment_method,
    COALESCE(o.payment_status, o.meta->>'payment_status') AS payment_status,
    COALESCE(o.payment_amount, o.total) AS payment_amount, -- Use total as fallback for payment_amount
    o.total
FROM
    public.orders AS o
LEFT JOIN
    public.drivers AS d ON o.assigned_to = d.id;

-- Grant permissions to the anon and authenticated roles if necessary
-- PO will handle RLS policies, but this is a common grant for views.
GRANT SELECT ON public.deliveries_for_admin TO anon, authenticated, service_role;

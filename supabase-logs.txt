-- ONE-PASTE CHECK: orders associated to a specific email
-- Change this email only:
with params as (
  select 'tabrez.ataher@gmail.com'::text as email
),
u as (
  -- find auth user (if exists)
  select id as user_id, email
  from auth.users
  where lower(email) = lower((select email from params))
  limit 1
),
p as (
  -- find profile (if exists)
  select user_id, email as profile_email
  from public.profiles
  where lower(email) = lower((select email from params))
  limit 1
),
candidates as (
  -- collect all orders that match by any route
  select
    o.*,
    (select email from params)              as target_email,
    lower(o.customer_email) = lower((select email from params))                as matched_customer_email_col,
    lower(o.customer_details->>'email') = lower((select email from params))    as matched_customer_details_json,
    case
      when o.customer_id is not null and o.customer_id = (select user_id from u) then true
      when o.customer_id is not null and o.customer_id = (select user_id from p) then true
      else false
    end as matched_by_customer_id
  from public.orders o
  where
       lower(o.customer_email) = lower((select email from params))
    or lower(o.customer_details->>'email') = lower((select email from params))
    or (o.customer_id is not null and o.customer_id in (
          select user_id from u
          union all
          select user_id from p
        ))
)
-- 1) quick headline counts
select
  (select email from params)                                  as target_email,
  count(*)                                                    as total_orders_for_email,
  count(*) filter (where matched_customer_email_col)          as by_customer_email_col,
  count(*) filter (where matched_customer_details_json)       as by_customer_details_json,
  count(*) filter (where matched_by_customer_id)              as by_customer_id_link,
  count(*) filter (where not matched_by_customer_id)          as missing_link_to_user_id
from candidates;

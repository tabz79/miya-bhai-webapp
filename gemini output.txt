  Part I: Database Schema & Migrations

  Four new migration files were created to structure the database.


  1.1. New Table: `coupons`
     File:* supabase/migrations/0004_create_coupons_table.sql
     Purpose:* Creates the primary table to store all coupon data and their
  attributes. It includes an index on the code for fast lookups and a trigger
  to automatically update the updated_at timestamp.
     Full Content:*

  `sql
      CREATE TABLE coupons (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        code TEXT NOT NULL UNIQUE,
        type TEXT NOT NULL CHECK (type IN ('percentage', 'flat')),
        value NUMERIC NOT NULL,
        is_active BOOLEAN DEFAULT true,
        max_uses INTEGER,
        uses_count INTEGER DEFAULT 0,
        starts_at TIMESTAMPTZ,
        expires_at TIMESTAMPTZ,
        created_at TIMESTAMPTZ DEFAULT NOW(),
        updated_at TIMESTAMPTZ DEFAULT NOW()
      );

      -- Create a function to update the updated_at timestamp
      CREATE OR REPLACE FUNCTION trigger_set_timestamp()
      RETURNS TRIGGER AS $$
      BEGIN
        NEW.updated_at = NOW();
        RETURN NEW;
      END;
      $$ LANGUAGE plpgsql;

      -- Create a trigger to automatically update updated_at on row change
      CREATE TRIGGER set_timestamp
      BEFORE UPDATE ON coupons
      FOR EACH ROW
      EXECUTE PROCEDURE trigger_set_timestamp();

      -- Add index on code for faster lookups
      CREATE INDEX idx_coupons_code ON coupons (UPPER(code));
      `


  1.2. Modification: `orders` Table
     File:* supabase/migrations/0005_add_coupon_to_orders.sql
     Purpose:* Alters the existing orders table to add columns for tracking
  which coupon was used and the discount amount.
     Full Content:*
      `sql
      ALTER TABLE orders
      ADD COLUMN coupon_code TEXT,
      ADD COLUMN discount_amount NUMERIC;
      `


  1.3. Data Seeding (Optional)

  1.4. New Database Function: `increment_coupon_uses`
     File:* supabase/migrations/0007_create_increment_coupon_function.sql
     Purpose:* Creates a stored procedure to safely increment the uses_count of
  a coupon. Using a database function ensures this operation is atomic and
  prevents race conditions where two orders could redeem the last available
  coupon simultaneously.
     Full Content:*

  `sql
      CREATE OR REPLACE FUNCTION increment_coupon_uses(coupon_id UUID)
      RETURNS void AS $$
        UPDATE coupons
        SET uses_count = uses_count + 1
        WHERE id = coupon_id;
      $$ LANGUAGE sql;
      `

  2.1. New Coupon Routes
     File:* src/routes/coupons.js
     Purpose:* This new file defines all API endpoints related to coupons,

     Full Content:*

  `javascript
      import express from 'express';
      import { supabase } from '../lib/supabase.js';
      import requireAdmin from '../middleware/requireAdmin.js';

      const router = express.Router();

      // Admin: Get all coupons
      router.get('/admin/coupons', requireAdmin, async (req, res) => {
        try {
          const { data, error } = await
  supabase.from('coupons').select('*').order('created_at', { ascending: false
  });
          if (error) throw error;
          res.json(data);
        } catch (err) {
          res.status(500).json({ error: err.message });
        }
      });

      // Admin: Create a new coupon
      router.post('/admin/coupons', requireAdmin, async (req, res) => {
        try {
          const { code, type, value, is_active, expires_at, max_uses } =
  req.body;
          const { data, error } = await supabase
            .from('coupons')
            .insert([{ code: code.toUpperCase(), type, value, is_active,
  expires_at, max_uses }])
            .select();
          if (error) throw error;
          res.status(201).json(data[0]);
        } catch (err) {
          res.status(500).json({ error: err.message });
        }
      });

      // Admin: Update a coupon
      router.put('/admin/coupons/:id', requireAdmin, async (req, res) => {
        try {
          const { id } = req.params;
          const { code, type, value, is_active, expires_at, max_uses } =
  req.body;
          const { data, error } = await supabase
            .from('coupons')
            .update({ code: code.toUpperCase(), type, value, is_active,
  expires_at, max_uses })
            .eq('id', id)
            .select();
          if (error) throw error;
          res.json(data[0]);
        } catch (err) {
          res.status(500).json({ error: err.message });
        }
      });

      // Admin: Delete a coupon
      router.delete('/admin/coupons/:id', requireAdmin, async (req, res) => {
        try {
          const { id } = req.params;
          const { error } = await supabase.from('coupons').delete().eq('id',
  id);
          if (error) throw error;
          res.status(204).send();
        } catch (err) {
          res.status(500).json({ error: err.message });
        }
      });

      // Public: Validate a coupon
      router.get('/coupons/validate/:code', async (req, res) => {
        try {
          const { code } = req.params;
          const { data: coupon, error } = await supabase
            .from('coupons')
            .select('*')
            .eq('UPPER(code)', code.toUpperCase())
            .single();

          if (error || !coupon) {
            return res.status(404).json({ error: 'Coupon not found' });
          }

          if (!coupon.is_active) {
            return res.status(400).json({ error: 'This coupon is not active'
  });
          }

          if (coupon.expires_at && new Date(coupon.expires_at) < new Date()) {
            return res.status(400).json({ error: 'This coupon has expired' });
          }

          if (coupon.max_uses && coupon.uses_count >= coupon.max_uses) {
            return res.status(400).json({ error: 'This coupon has reached its
  usage limit' });
          }

          res.json(coupon);
        } catch (err) {
          res.status(500).json({ error: err.message });
        }
      });

      export default router;
      `

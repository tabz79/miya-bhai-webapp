  +/**
  + * Defines a canonical order for menu categories to ensure
  consistent sorting across the UI.
  + * Categories not listed here will be appended alphabetically at        
  the end.
  + */
  +export const CANONICAL_CATEGORY_ORDER: string[] = [
  +  'Biryani',
  +  'Mandi',
  +  'Kebab',
  +  'Shawarma',
  +  'Starters',
  +  'Main Course',
  +  'Desserts',
  +  'Beverages',
  +];
  +
  +export const sortCategories = (categories: string[]): string[] =>       
   {
  +  return [...categories].sort((a, b) => {
  +    const indexA = CANONICAL_CATEGORY_ORDER.indexOf(a);
  +    const indexB = CANONICAL_CATEGORY_ORDER.indexOf(b);
  +
  +    if (indexA !== -1 && indexB !== -1) return indexA - indexB;
  // Both are in canonical order
  +    if (indexA !== -1) return -1; // A is canonical, B is not
  +    if (indexB !== -1) return 1;  // B is canonical, A is not
  +    return a.localeCompare(b); // Neither is canonical, sort
  alphabetically
  +  });
  +};
